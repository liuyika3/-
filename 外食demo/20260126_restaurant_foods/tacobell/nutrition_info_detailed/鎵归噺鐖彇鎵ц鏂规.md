from playwright.sync_api import sync_playwright
import json
import time

def scrape_all_details():
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        page = browser.new_page()
        page.goto('https://www.nutritionix.com/taco-bell/menu/premium')
        
        # 等待页面加载
        page.wait_for_selector('table')
        
        # 获取所有 [more info] 链接
        more_info_links = page.query_selector_all('a[href="viewLabel"]')
        
        all_details = []
        
        for idx, link in enumerate(more_info_links, 1):
            try:
                # 点击 [more info]
                link.click()
                
                # 等待对话框出现
                page.wait_for_selector('dialog[active]', timeout=5000)
                
                # 提取详细信息
                dialog = page.query_selector('dialog[active]')
                
                # 提取菜品名称
                name = dialog.query_selector('generic').inner_text()
                
                # 提取 Nutrition Facts
                nutrition_facts = {}
                # ... (提取逻辑)
                
                # 提取 Allergens
                allergens = {}
                allergen_rows = dialog.query_selector_all('table tr')
                for row in allergen_rows:
                    # ... (提取逻辑)
                
                # 提取 Ingredients
                ingredients = dialog.query_selector('paragraph').inner_text()
                
                detail = {
                    'name': name,
                    'nutrition_facts': nutrition_facts,
                    'allergens': allergens,
                    'ingredients': ingredients
                }
                
                all_details.append(detail)
                
                # 关闭对话框
                close_btn = dialog.query_selector('button')
                close_btn.click()
                
                # 等待对话框关闭
                time.sleep(0.5)
                
                print(f"[{idx}/466] {name} - 完成")
                
            except Exception as e:
                print(f"错误: {e}")
                continue
        
        # 保存结果
        with open('all_details.json', 'w', encoding='utf-8') as f:
            json.dump(all_details, f, ensure_ascii=False, indent=2)
        
        browser.close()

if __name__ == '__main__':
    scrape_all_details()
